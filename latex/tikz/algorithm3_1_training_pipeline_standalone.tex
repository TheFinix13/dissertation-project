\begin{algorithm}[H]
\caption{PPO high-frequency discrete trading loop with explicit vector state}
\label{alg:ppo_hft_explicit_state}

\Proc{\textsc{Train-Trading-Agent}($\{P_t\}$, $c$, $E$, $K$, $C_0$)}{
  \Comment{$\{P_t\}$: price series divided into $K$ daily episodes}\;
  \Comment{$c$: proportional transaction-fee coefficient}\;
  \Comment{$E$: training epochs; $C_0$: initial cash}\;
  Initialise policy parameters $\theta$ and value parameters $\phi$ randomly\;
  \For{epoch $e \in \{0,\ldots,E-1\}$}{
    Initialise trajectory buffer $\mathcal{B} \leftarrow \varnothing$\;
    \For{day $k \in \{1,\ldots,K\}$}{
      $\mathcal{B} \leftarrow
      \mathcal{B} \cup
      \Rollout(\{P_t^{(k)}\},\theta,c,C_0)$\;
    }
    Update $\theta$ by maximising the clipped PPO objective
    $L^{\mathrm{CLIP}}(\theta;\mathcal{B})$\;
    Update $\phi$ by minimising the value loss
    $L^{V}(\phi;\mathcal{B})$\;
  }
  \Return{$\theta^*,\phi^*$}\;
}

\BlankLine

\Proc{\Rollout{$\{P_t\}$, $\theta$, $c$, $C_0$}}{
  Initialise cash $C_0$, shares $n_0 \leftarrow 0$, and
  $s_0 \leftarrow [\Delta P_0,C_0,n_0]^\top$\;
  Initialise daily buffer $\mathcal{B}_{\mathrm{day}} \leftarrow \varnothing$\;
  \For{minute $t \in \{0,\ldots,T-1\}$}{
    \Comment{$T=390$ trading minutes}\;
    Compute valid-action mask $m_t$ from cash $C_t$ and shares $n_t$\;
    \Comment{Hold is always valid; Buy needs cash; Sell needs one share}\;
    Compute masked action probabilities
    $\widetilde{\pi}_\theta(a_t\mid s_t,m_t)$\;
    Sample $a_t\sim\widetilde{\pi}_\theta(a_t\mid s_t,m_t)$,
    where $a_t\in\{0{:}\mathrm{Hold},1{:}\mathrm{Buy},2{:}\mathrm{Sell}\}$\;
    Execute one-share transition:
    $\begin{cases}
      C_{t+1}=C_t,\quad n_{t+1}=n_t,
        & a_t=0,\\
      C_{t+1}=C_t-P_t(1+c),\quad n_{t+1}=n_t+1,
        & a_t=1,\\
      C_{t+1}=C_t+P_t(1-c),\quad n_{t+1}=n_t-1,
        & a_t=2;
    \end{cases}$\;
    Observe $\Delta P_{t+1}=(P_{t+1}-P_t)/P_t$\;
    Compute portfolio wealth $W_t=C_t+n_tP_t$ and reward
    $r_t=W_{t+1}-W_t$\;
    Construct
    $s_{t+1}\leftarrow[\Delta P_{t+1},C_{t+1},n_{t+1}]^\top$\;
    Store $(s_t,a_t,r_t,s_{t+1},m_t)$ in
    $\mathcal{B}_{\mathrm{day}}$\;
  }
  \Return{$\mathcal{B}_{\mathrm{day}}$}\;
}
\end{algorithm}

\vspace{0.35em}
\noindent\footnotesize
\textit{Interpretation.}
$C_t$ is cash, $n_t$ is the number of shares held, and
$W_t=C_t+n_tP_t$ is portfolio wealth. The action mask prevents infeasible
buys and sells. Transaction fees enter the cash transition, so they are
automatically reflected in the wealth-change reward.
