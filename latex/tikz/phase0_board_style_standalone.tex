\section*{Phase 0 --- same four beats on CartPole / Flappy}
\noindent\small
Dr Nguyen's board: \textbf{predict $\rightarrow$ score $\rightarrow$ differentiate $\rightarrow$ update}.
Games only change the environment. The training boards stay in that shape.
CartPole: $s\in\mathbb{R}^4$, $A=\{\mathrm{L},\mathrm{R}\}$, $r=+1$/step.
Flappy: $s\in\mathbb{R}^{12}$, $A=\{\mathrm{noop},\mathrm{flap}\}$.

\vspace{0.45em}
\noindent
\begin{minipage}[t]{0.32\textwidth}
\centering\textbf{His board}\par\vspace{0.25em}
\begin{Verbatim}[fontsize=\tiny,frame=single,framesep=3pt]
net = ResNet18()
opt = Adam(...)

# (x,y) fixed
preds = net(x)
loss  = MSE(preds,y)/N

opt.zero_grad()
loss.backward()
opt.step()
\end{Verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{0.32\textwidth}
\centering\textbf{REINFORCE (scratch)}\par\vspace{0.25em}
\begin{Verbatim}[fontsize=\tiny,frame=single,framesep=3pt]
policy = PolicyNet()
opt = Adam(policy)

# data by acting
traj = ROLLOUT(env, policy)
G = discounted_returns(traj)
L = -sum(log_pi * G)

opt.zero_grad()
L.backward()
opt.step()
\end{Verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{0.32\textwidth}
\centering\textbf{PPO (games = trading board)}\par\vspace{0.25em}
\begin{Verbatim}[fontsize=\tiny,frame=single,framesep=3pt]
policy = PolicyNet()
critic = ValueNet()
opt_pi, opt_V = Adam(), Adam()

B = ROLLOUT(env, policy)
A = advantages(B, critic)
L_pi = - PPO_CLIP(policy,B,A)
L_V  = MSE(V(s), returns)

opt_pi.zero_grad(); L_pi.backward(); opt_pi.step()
opt_V.zero_grad();  L_V.backward();  opt_V.step()
\end{Verbatim}
\end{minipage}

\vspace{0.55em}
\noindent
\begin{minipage}[t]{0.48\textwidth}
\centering\textbf{Tabular Q (binned CartPole only)}\par\vspace{0.25em}
\begin{Verbatim}[fontsize=\tiny,frame=single,framesep=3pt]
Q = zeros(bins, actions)

a = eps_greedy(Q[bin(s)])     # predict
s2, r, done = env.step(a)
y = r + gamma * max(Q[bin(s2)])  # score
Q[bin(s),a] += alpha*(y - Q[bin(s),a])  # update
\end{Verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\centering\textbf{DQN}\par\vspace{0.25em}
\begin{Verbatim}[fontsize=\tiny,frame=single,framesep=3pt]
Q, Q_bar = QNet(), copy(Q)
buf = ReplayBuffer()

a = eps_greedy(Q(s))          # predict
buf.add(s,a,r,s2,done)
y = r + gamma*max(Q_bar(s2))  # score
L = MSE(Q(s,a), y)
opt.zero_grad(); L.backward(); opt.step()
\end{Verbatim}
\end{minipage}

\vspace{0.5em}
\noindent\textbf{Key translation (games Phase 0)}
\begin{itemize}\setlength\itemsep{0.1em}
  \item Labels $y$ become rewards $r$ from \texttt{env.step}.
  \item Fixed dataset becomes a rollout that depends on $\pi_\theta$.
  \item Same PPO board as trading: only \texttt{ROLLOUT(env)} vs \texttt{ROLLOUT(prices)}.
  \item Tabular Q needs binning --- reason we do not use it for stocks.
\end{itemize}
