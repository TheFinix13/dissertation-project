\section*{Phase 0 pseudocode --- games + algorithms (simple-modelling track)}
\small
CartPole and Flappy Bird share the same training procedures; only
\texttt{RESET}/\texttt{STEP}/reward differ. REINFORCE and tabular Q-learning
are implemented from scratch in the repository; A2C/PPO/DQN use the same
update mathematics via Stable-Baselines3.

\vspace{0.4em}
\noindent\textbf{CartPole-v1:}
$s\in\mathbb{R}^4$ (pos, vel, angle, ang-vel);
$A=\{0{:}\mathrm{LEFT},1{:}\mathrm{RIGHT}\}$;
$r=+1$/step upright; solved $\approx 475$.\\
\noindent\textbf{FlappyBird-v0:}
$s\in\mathbb{R}^{12}$;
$A=\{0{:}\mathrm{noop},1{:}\mathrm{FLAP}\}$;
$r=+0.1$ alive, $+1$ pipe, $-1$ death.

\renewcommand{\thealgocf}{P0.1}
\begin{algorithm}[H]
\caption{REINFORCE (Monte Carlo policy gradient) --- coded from scratch}
\Proc{\textsc{Train-REINFORCE}(env, $E$, $\gamma$, $\eta$)}{
  Initialise policy parameters $\theta$\;
  \For{episode $e \in \{1,\ldots,E\}$}{
    $\tau \leftarrow \Rollout(\mathrm{env},\pi_\theta)$\;
    \Comment{Discounted returns from the end of the episode}\;
    $G_T \leftarrow 0$\;
    \For{$t = T-1,\ldots,0$}{
      $G_t \leftarrow r_t + \gamma G_{t+1}$\;
    }
    Optionally normalise $\{G_t\}$ to zero mean / unit variance\;
    $L(\theta) \leftarrow -\sum_t \log\pi_\theta(a_t\mid s_t)\,G_t$\;
    $\theta \leftarrow \theta - \eta\nabla_\theta L(\theta)$\;
  }
  \Return{$\theta$}\;
}
\end{algorithm}

\renewcommand{\thealgocf}{P0.2}
\begin{algorithm}[H]
\caption{Tabular Q-learning on \emph{discretized} CartPole}
\Proc{\textsc{Train-Tabular-Q}(env, $E$, $\alpha$, $\gamma$, $\varepsilon$)}{
  Initialise table $Q[\mathrm{bin}(s),a]\leftarrow 0$\;
  \Comment{Continuous $s$ must be binned --- does not scale to trading states}\;
  \For{episode $e \in \{1,\ldots,E\}$}{
    $s\leftarrow\mathrm{env.RESET}()$\;
    \While{not done}{
      With prob.\ $\varepsilon$: $a\leftarrow\mathrm{Uniform}(A)$;
      else $a\leftarrow\arg\max_{a'}Q[\mathrm{bin}(s),a']$\;
      $s',r,\mathrm{done}\leftarrow\mathrm{env.STEP}(a)$\;
      $y\leftarrow r+\gamma\max_{a'}Q[\mathrm{bin}(s'),a']\cdot(1-\mathrm{done})$\;
      $Q[\mathrm{bin}(s),a]\leftarrow Q[\mathrm{bin}(s),a]+\alpha\bigl(y-Q[\mathrm{bin}(s),a]\bigr)$\;
      $s\leftarrow s'$\;
    }
  }
  \Return{$Q$}\;
}
\end{algorithm}

\renewcommand{\thealgocf}{P0.3}
\begin{algorithm}[H]
\caption{PPO clipped update (carried into Phase~1 trading)}
\Proc{\textsc{Train-PPO}(env, steps, $\gamma$, $\varepsilon_{\mathrm{clip}}$, $K$)}{
  Initialise actor $\pi_\theta$ and critic $V_\phi$\;
  \While{total steps $<$ budget}{
    Collect batch $\mathcal{B}$ with $\pi_\theta$; store also $\log\pi_{\theta_{\mathrm{old}}}$\;
    Compute advantages $A_t$\;
    \For{epoch $k \in \{1,\ldots,K\}$}{
      $\rho_t\leftarrow\pi_\theta(a_t\mid s_t)/\pi_{\theta_{\mathrm{old}}}(a_t\mid s_t)$\;
      $L^{\mathrm{CLIP}}(\theta)\leftarrow -\mathrm{mean}_t\min\bigl(\rho_t A_t,\;
        \mathrm{clip}(\rho_t,1\pm\varepsilon_{\mathrm{clip}})A_t\bigr)$\;
      $L^{V}(\phi)\leftarrow\mathrm{mean}_t\bigl(V_\phi(s_t)-G_t\bigr)^2$\;
      Update $\theta,\phi$ on $L^{\mathrm{CLIP}}+c_v L^{V}$\;
    }
    $\theta_{\mathrm{old}}\leftarrow\theta$\;
  }
  \Return{$\theta,\phi$}\;
}
\end{algorithm}

\vspace{0.3em}
\noindent\textit{DQN:} same Bellman target as tabular Q with network $Q_\phi$,
replay buffer, and frozen target net $Q_{\bar\phi}$.
\textit{A2C:} actor loss $-\log\pi_\theta(a|s)A_t$ with $A_t=G_t-V_\phi(s)$;
no clip. Suite: run Random / REINFORCE / A2C / PPO / DQN on each env\_id.
