\begin{algorithm}[H]
\caption{High-frequency baseline discrete trading agent (policy gradient)}
\label{alg:baseline_discrete}

\Proc{\textsc{Train-Discrete-Policy}($\{P_t\}$, $c$, $E$, $K$)}{
  \Comment{$\{P_t\}$: historical price series}\;
  \Comment{$c$: transaction-fee coefficient}\;
  \Comment{$E$: number of training epochs}\;
  \Comment{$K$: number of trading-day episodes per epoch}\;
  Initialise policy parameters $\theta$ and value parameters $\phi$ randomly\;
  \For{epoch $e = 1$ \KwTo $E$}{
    \For{each trading day (episode) $k = 1$ \KwTo $K$}{
      Initialise state $s_0 = [\Delta P_0,\; h_0{=}0]^\top$\;
      Initialise trajectory buffer $\mathcal{B} = \{\}$\;
      \For{minute $t = 0$ \KwTo $T-1$}{
        \Comment{$T = 390$ minutes in a US equity session}\;
        Forward pass: compute $\pi_\theta(a_t \mid s_t)$\;
        Sample $a_t \sim \pi_\theta(a_t \mid s_t)$\;
        \Comment{$a_t \in \{0{:}\,\mathrm{Hold},\; 1{:}\,\mathrm{Buy},\; 2{:}\,\mathrm{Sell}\}$}\;
        \uIf{$a_t = 1$}{
          $h_{t+1} \leftarrow 1$\;
          \Comment{Buy / Long}\;
        }
        \uElseIf{$a_t = 2$}{
          $h_{t+1} \leftarrow 0$\;
          \Comment{Sell / Cash}\;
        }
        \Else{
          $h_{t+1} \leftarrow h_t$\;
          \Comment{Hold}\;
        }
        Observe $\Delta P_{t+1} = (P_{t+1} - P_t)/P_t$\;
        Calculate reward $r_t = (h_t \cdot \Delta P_{t+1}) - c \cdot |h_{t+1} - h_t|$\;
        Construct $s_{t+1} = [\Delta P_{t+1},\; h_{t+1}]^\top$\;
        Store $(s_t, a_t, r_t, s_{t+1})$ in $\mathcal{B}$\;
      }
    }
    Update $\theta$ by policy-gradient ascent on $\mathcal{B}$\;
    \Comment{PPO is one such update rule; treat as black-box optimiser}\;
    Update value parameters $\phi$\;
  }
  \Return{$\theta^*$}\;
}
\end{algorithm}

\vspace{0.6em}
\noindent\small\textit{Note for the reader.} This is Iteration~1 only: three discrete actions, binary holding $h_t\in\{0,1\}$, one stock, one day $=$ one trajectory. Later iterations expand the action space; they are not claimed here.
